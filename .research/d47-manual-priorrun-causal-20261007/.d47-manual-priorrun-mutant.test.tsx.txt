it('keeps a concurrent manual response out of the saved replay result', async () => {
  vi.useFakeTimers();
  let currentTitle = 'Brooklyn Bowl';
  const cdpEvents = new Set<(source: { tabId: number }, method: string, params: object) => void>();
  const emit = (method: string, params: object) => {
    for (const fn of cdpEvents) fn({ tabId: 37 }, method, params);
  };
  let binding = '';
  let releaseReload!: () => void;
  const reloadGate = new Promise<void>((resolve) => {
    releaseReload = resolve;
  });
  h.acquire.mockResolvedValue({ send: h.send, release: h.release });
  h.release.mockResolvedValue(undefined);
  h.send.mockImplementation(async (method: string, params: Record<string, any>) => {
    if (method === 'Page.getFrameTree')
      return {
        frameTree: { frame: { id: 'main-frame', url: h.tab.url, loaderId: 'original-loader' } },
      };
    if (method === 'Runtime.enable')
      emit('Runtime.executionContextCreated', {
        context: { id: 1, uniqueId: 'prior', auxData: { frameId: 'main-frame', isDefault: true } },
      });
    if (method === 'Runtime.addBinding') binding = params.name;
    if (method === 'Page.addScriptToEvaluateOnNewDocument') return { identifier: 'replay-script' };
    if (method === 'Page.reload') {
      await reloadGate;
      emit('Page.frameStartedNavigating', {
        frameId: 'main-frame',
        loaderId: 'new-loader',
        navigationType: 'reload',
        url: h.tab.url,
      });
      emit('Runtime.executionContextCreated', {
        context: {
          id: 2,
          uniqueId: 'current',
          auxData: { frameId: 'main-frame', isDefault: true },
        },
      });
      emit('Runtime.bindingCalled', {
        name: binding,
        executionContextId: 2,
        payload: JSON.stringify({
          source: 'fetch',
          method: 'GET',
          url: 'https://electronic.vegas/api/events',
          body: JSON.stringify({ events: [{ title: currentTitle }] }),
          request_body_key: 'none',
          request_sequence: 1,
          status: 200,
          ts_ms: 1,
          body_size: 37,
          body_truncated: false,
        }),
      });
      emit('Page.frameNavigated', {
        frame: { id: 'main-frame', url: h.tab.url, loaderId: 'new-loader' },
      });
      emit('Runtime.bindingCalled', {
        name: binding,
        executionContextId: 2,
        payload: JSON.stringify({ __matrx_capture_hook: 'network-tap', nonce: 'a'.repeat(32) }),
      });
    }
    return {};
  });
  const runtimeMessage = vi.fn((envelope: { kind: string; payload: unknown }) => {
    // Chrome delivers this content-script message to the registered SW handler.
    for (const callback of h.listeners.get(envelope.kind) ?? [])
      callback(envelope.payload, { tab: { id: 37 }, documentId: 'manual-document' });
    return Promise.resolve({ ack: true });
  });
  Object.assign(chrome, {
    runtime: {
      id: 'extension',
      sendMessage: runtimeMessage,
      onMessage: { addListener: vi.fn(), removeListener: vi.fn() },
      onConnect: { addListener: vi.fn() },
    },
    alarms: { onAlarm: { addListener: vi.fn() } },
    tabs: { onUpdated: { addListener: vi.fn() } },
    scripting: {
      executeScript: vi.fn(async () => [{ documentId: 'replayed-document', result: true }]),
    },
    debugger: {
      sendCommand: vi.fn(),
      onEvent: {
        addListener: (fn: (source: { tabId: number }, method: string, params: object) => void) =>
          cdpEvents.add(fn),
        removeListener: (fn: (source: { tabId: number }, method: string, params: object) => void) =>
          cdpEvents.delete(fn),
      },
      onDetach: { addListener: vi.fn(), removeListener: vi.fn() },
    },
  });
  bootstrapBackground();
  const hook = renderHook(useNetworkCapture);
  await act(async () => hook.result.current.start());
  networkRelayIsolated();
  const config = {
    url_filter: 'https://electronic.vegas/api/events',
    method: 'GET',
    key_path: 'events',
    body_match: 'ignore',
  };
  const runReplay = () =>
    runNetworkCapturePattern(config, 37, { initiation: 'user', timeoutMs: 100 });
  const replay = runReplay();
  for (let i = 0; i < 100 && !h.send.mock.calls.some(([method]) => method === 'Page.reload'); i++)
    await Promise.resolve();
  expect(h.send.mock.calls.some(([method]) => method === 'Page.reload')).toBe(true);
  await act(async () => {
    window.dispatchEvent(
      new MessageEvent('message', {
        source: window,
        data: {
          __matrx_net: true,
          event: {
            source: 'fetch',
            method: 'GET',
            url: 'https://electronic.vegas/api/events',
            body: '{"events":[{"title":"Manual prior response"}]}',
            request_body_key: 'none',
            request_sequence: 999,
            status: 200,
            ts_ms: 2,
            body_size: 49,
            body_truncated: false,
          },
        },
      }),
    );
  });
  expect(runtimeMessage).toHaveBeenCalledWith(
    expect.objectContaining({ kind: CHANNELS.NET_CAPTURE_EVENT }),
  );
  expect(hook.result.current.events.map((event) => event.body)).toEqual([
    '{"events":[{"title":"Manual prior response"}]}',
  ]);
  releaseReload();
  for (let i = 0; i < 20; i++) await Promise.resolve();
  await vi.advanceTimersByTimeAsync(100);
  expect(await replay).toEqual([{ title: 'Brooklyn Bowl' }]);
  currentTitle = 'The Fonda Theatre';
  const second = runReplay();
  for (
    let i = 0;
    i < 100 && h.send.mock.calls.filter(([method]) => method === 'Page.reload').length < 2;
    i++
  )
    await Promise.resolve();
  expect(h.send.mock.calls.filter(([method]) => method === 'Page.reload')).toHaveLength(2);
  await vi.advanceTimersByTimeAsync(100);
  expect(await second).toEqual([{ title: 'The Fonda Theatre' }]);
});
